import { useState } from 'react';
import { Button } from '@components/ui';
import { InviteMembersDialog } from '@blocks/invite-members-dialog';

export const ROLES = [
  { value: 'member', label: 'Member' },
  { value: 'admin', label: 'Admin' },
  { value: 'viewer', label: 'Viewer' },
];

export const fakeInvite = async (invites) => {
  // A pretend server: anyone at taken.com is already a member.
  await new Promise((resolve) => setTimeout(resolve, 800));
  const taken = invites.filter((invite) => invite.email.endsWith('@taken.com'));
  return { failed: Object.fromEntries(taken.map((invite) => [invite.email, 'Already a member'])) };
};

export const InviteDemo = () => {
  const [open, setOpen] = useState(false);

  const handleOpen = () => setOpen(true);

  return (
    <>
      <Button onClick={handleOpen}>Invite people</Button>
      <InviteMembersDialog open={open} onOpenChange={setOpen} roles={ROLES} onInvite={fakeInvite} />
    </>
  );
};

## Invite by email

Type or paste addresses — a comma, a space or Enter ends each one — pick a role and send. Try `sam@taken.com`: the pretend server refuses it, so it stays in the dialog with its reason while the others are sent.

## Add it

```bash
pnpm kit:add invite-members-dialog
```

It lands in `src/blocks/invite-members-dialog/` with its English and Arabic copy; `pnpm kit:update` keeps it current and `pnpm kit:remove invite-members-dialog` takes it out again.

The block never calls an API itself: `onInvite` is where your app sends the invitations.

## Accessibility

- The addresses field is labelled and described by its hint or error; each chip's remove button is named after its address, and an address that isn't valid says so to screen readers too, not only in red.
- Errors from the server appear in an alert, which is announced; while sending, the form is disabled and the send button shows it is busy.

## RTL notes

- Addresses are isolated (`<bdi>`), so a Latin address reads correctly inside Arabic text; chips wrap from the start of the field.
