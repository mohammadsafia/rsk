import { useState } from 'react';
import { CheckboxGroup } from '@components/ui';
import { Check } from 'lucide-react';

export const OPTIONS = ['Email', 'SMS', 'Push'];

export const ChannelsDemo = () => {
  const [selected, setSelected] = useState(['Email']);
  const toggle = (option, checked) => setSelected((current) => (checked ? [...current, option] : current.filter((value) => value !== option)));

  return (
    <CheckboxGroup aria-label="Notification channels" className="w-full max-w-sm">
      {OPTIONS.map((option) => (
        <CheckboxGroup.Item key={option} checked={selected.includes(option)} onCheckedChange={(checked) => toggle(option, !!checked)}>
          <span className="flex-1">{option}</span>
          <CheckboxGroup.Indicator>
            <Check className="hidden size-3 group-data-[state=checked]/checkbox-item:block" strokeWidth={3} />
          </CheckboxGroup.Indicator>
        </CheckboxGroup.Item>
      ))}
    </CheckboxGroup>
  );
};

## Option cards

Each item is a whole-card checkbox; checked cards take the accent border and tint.

## Accessibility

- The container has `role="group"`; give it an `aria-label` or `aria-labelledby` naming the choice.
- Each card is a `role="checkbox"` button with `aria-checked`, so the text inside is its accessible name.

## RTL notes

- The label sits at the start and the indicator at the end, so they swap sides in Arabic.
