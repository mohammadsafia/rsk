import { Button, Toast } from '@components/ui';
import { useToast } from '@hooks/shared';

export const TonesDemo = () => {
  const { toast } = useToast();
  return (
    <div className="flex flex-wrap gap-2">
      <Button variant="outline" onClick={() => toast({ title: 'Draft saved', description: 'You can finish it later.' })}>Neutral</Button>
      <Button variant="outline" onClick={() => toast({ tone: 'success', title: 'Invoice paid', description: 'The payment was received.' })}>Success</Button>
      <Button variant="outline" onClick={() => toast({ tone: 'warning', title: 'Session expiring', description: 'Save your work.' })}>Warning</Button>
      <Button variant="outline" onClick={() => toast({ tone: 'danger', title: 'Upload failed', description: 'The file is larger than 10 MB.' })}>Danger</Button>
      <Button variant="outline" onClick={() => toast({ tone: 'info', title: 'New results', description: '3 lab results arrived.' })}>Info</Button>
    </div>
  );
};

export const ActionDemo = () => {
  const { toast } = useToast();
  return (
    <Button
      variant="outline"
      onClick={() =>
        toast({
          title: 'Visit cancelled',
          description: 'Layla Haddad · 10:30',
          action: <Toast.Action altText="Undo cancellation">Undo</Toast.Action>,
        })
      }
    >
      Cancel visit
    </Button>
  );
};

## Tones

Call `toast()` from `useToast`. The tone colours only the icon; the surface stays neutral.

<Example

toast({ tone: 'danger', title: 'Upload failed', description: 'The file is larger than 10 MB.' });`}

</Example>

`toast()` returns `{ id, dismiss, update }`. Toasts close after 5 seconds.

## Accessibility

- The viewport is a labelled region; new toasts are announced politely, and F8 moves focus to it.
- Don't put the only way to do something in a toast; it disappears.

## RTL notes

- Toasts appear in the top-end corner (top-left in Arabic) and swipe away toward the end.
