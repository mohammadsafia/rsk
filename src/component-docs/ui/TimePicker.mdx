import { useState } from 'react';
import { TimePicker } from '@components/ui';

export const ControlledDemo = () => {
  const [time, setTime] = useState('09:30 AM');
  return (
    <div className="grid w-60 gap-1.5">
      <label htmlFor="visit-time" className="text-fg text-sm font-medium">Visit time</label>
      <TimePicker id="visit-time" name="visit-time" step={15} value={time} onChange={(value) => setTime(value ?? null)} />
      <p className="text-fg-muted text-xs">Value: <span dir="ltr" className="font-mono">{time ?? '—'}</span></p>
    </div>
  );
};

## Basic

Pick the hour, minute and period; Confirm applies it, and the clear control empties the field.

## Accessibility

- Label it with `id` and a `<label htmlFor>` (or use the form field); `aria-describedby` and `aria-required` pass through.
- The hour, minute and AM/PM columns are keyboard-reachable listboxes.

## RTL notes

- The control follows the page direction, while the time value itself stays left to right (`09:30 AM`) with Western digits.
