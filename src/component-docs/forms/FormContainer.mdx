import { demoSource } from '@component-docs/blocks';
import { useForm } from 'react-hook-form';
import { FormBasicDemo } from '@pages/components/components.data.demos';
import { FormContainer, FormInput } from '@components/forms';

export const ActionsDemo = () => {
  const form = useForm({ defaultValues: { name: 'Layla Haddad' } });
  return (
    <FormContainer formContext={form} onSuccess={() => new Promise((resolve) => setTimeout(resolve, 800))} className="grid w-96 gap-3">
      <FormInput name="name" label="Name" />
      <FormContainer.Actions confirmOnCancel onCancel={() => form.reset()} />
    </FormContainer>
  );
};

## Basic

Pass a `useForm` result as `formContext`; `onSuccess` runs with the values after validation passes.

</Example>

## Actions

`FormContainer.Actions` renders the footer row: your extra actions, then Cancel and Save. Save shows a spinner while `onSuccess` runs. Edit the name, then press Cancel to see the confirmation.

## Accessibility

- A native `<form>`: Enter in a field submits, and Save is a `type="submit"` button with `aria-busy` while saving.
- On a failed submit, react-hook-form moves focus to the first invalid field (its default `shouldFocusError`).

## RTL notes

- The actions row aligns to the end, so Save sits on the left in Arabic, with Cancel before it.
