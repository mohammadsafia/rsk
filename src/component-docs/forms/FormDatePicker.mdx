import { useForm } from 'react-hook-form';
import { FormContainer, FormDatePicker } from '@components/forms';

export const DateDemo = () => {
  const form = useForm({ defaultValues: { visit: new Date(2026, 9, 16), birth: undefined } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormDatePicker name="visit" control={form.control} label="Visit date" disabled={{ before: new Date(2026, 9, 1) }} />
      <FormDatePicker name="birth" control={form.control} label="Date of birth" dateFormat="dd/MM/yyyy" placeholder="dd/mm/yyyy" required />
    </FormContainer>
  );
};

## Typed or picked

Type digits and the mask adds the separators, or open the calendar. The value stored in the form is a `Date`.

## Accessibility

- The text input is labelled and uses a numeric keyboard on mobile; the calendar button is labelled "Open calendar".
- Keyboard users can type the date without opening the calendar.

## RTL notes

- The calendar opens in Arabic with right-to-left weeks; the typed date stays in its mask order with Western digits.
