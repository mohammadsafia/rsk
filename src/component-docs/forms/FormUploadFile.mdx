import { useForm } from 'react-hook-form';
import { FormContainer, FormFileUploader } from '@components/forms';

export const SingleDemo = () => {
  const form = useForm({ defaultValues: { referral: null } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormFileUploader name="referral" control={form.control} label="Referral letter" accept=".pdf,.doc,.docx" placeholder="Pick a document" />
    </FormContainer>
  );
};

export const MultipleDemo = () => {
  const form = useForm({ defaultValues: { attachments: [], labels: [] } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-4">
      <FormFileUploader name="attachments" control={form.control} label="Attachments (list)" accept=".pdf,.txt" multiple />
      <FormFileUploader name="labels" control={form.control} label="Attachments (chips)" multiple listVariant="chips" />
    </FormContainer>
  );
};

export const PreviewDemo = () => {
  const form = useForm({ defaultValues: { photo: null } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormFileUploader name="photo" control={form.control} label="Profile photo" accept="image/*" showPreview />
    </FormContainer>
  );
};

## Single file

The value is a `File` (or `null`). Drop a file on the zone or click to browse.

<Example
  title="One document"
  code={`<FormFileUploader
  name="referral"
  control={form.control}
  label="Referral letter"
  accept=".pdf,.doc,.docx"
  placeholder="Pick a document"
/>`}
>
  <SingleDemo />
</Example>

## Multiple files

With `multiple`, the value is `File[]`. Picked files show as a list (default) or as chips.

<Example
  title="List and chips"
  code={`<FormFileUploader name="attachments" control={form.control} label="Attachments" accept=".pdf,.txt" multiple />
<FormFileUploader name="labels" control={form.control} label="Attachments" multiple listVariant="chips" />`}
>
  <MultipleDemo />
</Example>

## Accessibility

- The drop zone wraps a real file input, so it is keyboard-operable and labelled by the field label.
- Each picked file has a labelled remove button.

## RTL notes

- File sizes are isolated left to right (`2.4 MB`); the remove button sits at the end of each row.
