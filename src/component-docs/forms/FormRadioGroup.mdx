import { useForm } from 'react-hook-form';
import { FormContainer, FormRadioGroup } from '@components/forms';

export const VISIT_TYPES = [
  { id: 'in-person', name: 'In person' },
  { id: 'video', name: 'Video call' },
  { id: 'phone', name: 'Phone' },
];

export const RadioDemo = () => {
  const form = useForm({ defaultValues: { visitType: 'in-person' } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-full max-w-lg gap-3">
      <FormRadioGroup
        name="visitType"
        control={form.control}
        label="Visit type"
        options={VISIT_TYPES}
        getOptionLabel={(option) => option.name}
        getOptionValue={(option) => option.id}
        required
      />
    </FormContainer>
  );
};

## Accessibility

- A `radiogroup` labelled by the field label; Arrow keys move and select, and errors are linked with `aria-describedby`.

## RTL notes

- Cards run in reading order; Arrow keys follow the reading direction.
