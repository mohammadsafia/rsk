import { useForm } from 'react-hook-form';
import { FormContainer, FormDateRange } from '@components/forms';

export const RangeDemo = () => {
  const form = useForm({ defaultValues: { stay: undefined, admittedOn: undefined, dischargedOn: undefined } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormDateRange name="stay" control={form.control} label="Stay" placeholder="Select dates" />
      <FormDateRange fromName="admittedOn" toName="dischargedOn" control={form.control} label="Admission" required />
    </FormContainer>
  );
};

## One value or two fields

With `name`, the form stores `{ from, to }`. With `fromName` and `toName`, it writes two separate fields, which suits APIs that expect `startDate` and `endDate`.

## Accessibility

- One labelled text input takes both dates with a mask; the calendar button opens a two-month range picker.

## RTL notes

- The calendar runs right to left in Arabic; the typed range keeps its order with Western digits.
