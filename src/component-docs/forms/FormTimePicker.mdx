import { useForm } from 'react-hook-form';
import { FormContainer, FormTimePicker } from '@components/forms';

export const TimeDemo = () => {
  const form = useForm({ defaultValues: { start: '09:30:00', reminder: '' } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormTimePicker name="start" control={form.control} label="Start time" step={15} valueFormat="HH:mm:ss" description="Stored as 09:30:00" />
      <FormTimePicker name="reminder" control={form.control} label="Reminder" use24hFormat min="08:00" max="18:00" step={30} />
    </FormContainer>
  );
};

## Display and stored formats

The field shows `9:30 AM` (or `09:30` with `use24hFormat`). `valueFormat` makes the form store the API's format instead, here `HH:mm:ss`.

## Accessibility

- The trigger is labelled by the field label; hour, minute and AM/PM are keyboard-reachable listboxes.

## RTL notes

- The time itself stays left to right with Western digits in both languages.
