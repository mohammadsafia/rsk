import { useForm } from 'react-hook-form';
import { FormChipInput, FormContainer } from '@components/forms';

export const isEmail = (value) => /^[^\s@]+@[^\s@.]+\.[^\s@]+$/.test(value);

export const RecipientsDemo = () => {
  const form = useForm({ defaultValues: { cc: ['lina@example.com'] } });

  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormChipInput
        name="cc"
        label="CC"
        description="Separate addresses with a comma, or press Enter after each."
        placeholder="name@example.com"
        inputMode="email"
        validate={isEmail}
        invalidLabel="(not a valid email address)"
      />
    </FormContainer>
  );
};

## Recipients

The value is the list of chips (`string[]`). Validate the list in your schema or `rules`; `validate` marks the chips themselves.

## Accessibility

- The label names the input and the list of chips; the error or help text describes the input.

## RTL notes

- As ChipInput: chips flow from the start and each item is isolated, so addresses read correctly in Arabic.
