import { useForm } from 'react-hook-form';
import { FormContainer, FormTextarea } from '@components/forms';

export const TextareaDemo = () => {
  const form = useForm({ defaultValues: { notes: '' } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormTextarea name="notes" label="Visit notes" description="Visible to the care team only" placeholder="Symptoms, findings, plan…" />
    </FormContainer>
  );
};

## Accessibility

- The label and the message line are wired to the textarea; errors set `aria-invalid`.

## RTL notes

- Text follows the page direction; Arabic notes read right to left.
