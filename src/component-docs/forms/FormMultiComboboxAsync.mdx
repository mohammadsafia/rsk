import { useForm } from 'react-hook-form';
import { fetchDoctors } from '@component-docs/blocks';
import { FormContainer, FormMultiCombobox } from '@components/forms';

export const AsyncMultiDemo = () => {
  const form = useForm({ defaultValues: { careTeam: [] } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormMultiCombobox
        name="careTeam"
        control={form.control}
        label="Care team"
        options={fetchDoctors}
        getOptionLabel={(option) => option?.name ?? ''}
        getOptionValue={(option) => option?.id ?? ''}
        placeholder="Add doctors…"
      />
    </FormContainer>
  );
};

## Loading from an API

Same fetcher contract as the [single async combobox](/docs/components/form-combobox-async#the-fetcher-contract). Every selected id is sent as `selectedItemsIds`, so the API can return them even when they aren't on the current page.

## Accessibility

- A loading row shows while a page loads, and "Loading more…" while the next page arrives; selected options keep their badges even when they scroll out of the list.

## RTL notes

- Badges run in reading order; Arabic search terms are sent as typed.
