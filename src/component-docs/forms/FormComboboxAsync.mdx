import { useForm } from 'react-hook-form';
import { fetchDoctors } from '@component-docs/blocks';
import { FormCombobox, FormContainer } from '@components/forms';

export const AsyncDemo = () => {
  const form = useForm({ defaultValues: { doctorId: '' } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormCombobox
        name="doctorId"
        control={form.control}
        label="Doctor"
        options={fetchDoctors}
        getOptionLabel={(option) => option?.name ?? ''}
        getOptionValue={(option) => option?.id ?? ''}
        placeholder="Search doctors…"
      />
    </FormContainer>
  );
};

## Loading from an API

Pass a fetcher as `options`. It receives a query string with `pageNumber`, `pageSize`, `searchTerm` and your `urlSearchParams`, and returns a page. Scrolling to the end loads the next page; typing is debounced.

The docs have no backend, so this example uses a local mock that answers like the API.

## The fetcher contract

```ts
type AsyncOptionsFn<TOption> = (queryParams: string) => Promise<TOption[] | PaginatedDataTable<TOption>>;
// queryParams: "pageNumber=1&pageSize=20&searchTerm=had&selectedItemsIds=dr-3&…your urlSearchParams"
```

Return `{ data, pagination: { page, pageSize, total, totalPages } }` for paging, or a plain array for a single page. The current value's id is sent as `selectedItemsIds`, so the API can include it even when it isn't on the first page.

## Accessibility

- While a page loads, the list shows a loading row; "No results" is shown when the search matches nothing.

## RTL notes

- Search terms are sent as typed; Arabic searches work the same as English ones.
