import { useForm } from 'react-hook-form';
import { FormCheckbox, FormContainer } from '@components/forms';

export const ConsentDemo = () => {
  const form = useForm({ defaultValues: { reminders: true, terms: false } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormCheckbox name="reminders" control={form.control} label="Send visit reminders by SMS" />
      <FormCheckbox name="terms" control={form.control} label="I agree to the privacy policy" required rules={{ required: 'Accept the policy to continue' }} />
    </FormContainer>
  );
};

## Accessibility

- The whole card is the label, so clicking the text toggles the box; errors are linked with `aria-describedby`.

## RTL notes

- The box sits at the start of the card, on the right in Arabic.
