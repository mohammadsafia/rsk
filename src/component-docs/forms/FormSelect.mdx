import { useForm } from 'react-hook-form';
import { FormContainer, FormSelect } from '@components/forms';

export const ROLES = [
  { id: 'admin', name: 'Admin' },
  { id: 'doctor', name: 'Doctor' },
  { id: 'reception', name: 'Reception' },
];

export const SelectDemo = () => {
  const form = useForm({ defaultValues: { role: 'doctor' } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormSelect
        name="role"
        label="Role"
        control={form.control}
        options={ROLES}
        getOptionLabel={(option) => option.name}
        getOptionValue={(option) => option.id}
        placeholder="Select a role"
        required
      />
    </FormContainer>
  );
};

## Options

Options can be strings or objects; `getOptionLabel` and `getOptionValue` read them. A ✕ in the trigger empties the field unless `clearable={false}`.

## Accessibility

- The trigger is a labelled `combobox` that opens a `listbox`.
- The clear ✕ is a pointer shortcut inside the trigger and is not keyboard-focusable; use `clearable={false}` with an explicit empty option if keyboard clearing matters.

## RTL notes

- The chevron and clear ✕ sit at the end of the field, on the left in Arabic.
