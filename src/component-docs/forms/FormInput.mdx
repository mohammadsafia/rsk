import { useForm } from 'react-hook-form';
import { FormContainer, FormInput } from '@components/forms';

export const InputDemo = () => {
  const form = useForm({ defaultValues: { name: '', email: '', mrn: 'MRN-' } });
  return (
    <FormContainer formContext={form} onSuccess={() => {}} className="grid w-96 gap-3">
      <FormInput name="name" label="Full name" required rules={{ required: 'Enter the patient’s name' }} />
      <FormInput name="email" type="email" label="Email" description="Used for visit reminders" />
      <FormInput name="mrn" label="Record number" tooltip="Printed on the patient card" />
    </FormContainer>
  );
};

## Fields

`FormInput` registers itself with the surrounding `FormContainer`, so it needs no `control`.

## Accessibility

- The label is linked to the input, and the description or error is announced through `aria-describedby`.
- Set `type` and `autoComplete` (`email`, `tel`, `name`) so browsers and assistive tech can help fill the field.

## RTL notes

- The input uses `dir="auto"`, so an email typed into an Arabic form still reads left to right.
