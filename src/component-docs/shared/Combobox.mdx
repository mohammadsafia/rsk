import { useState } from 'react';
import { Combobox } from '@components/shared';
import { Conditional } from '@components/utils';
import { CheckIcon } from 'lucide-react';

export const DOCTORS = [
  { id: 'lina', name: 'Dr. Lina Khoury', specialty: 'Cardiology' },
  { id: 'omar', name: 'Dr. Omar Nasser', specialty: 'Dermatology' },
  { id: 'sara', name: 'Dr. Sara Haddad', specialty: 'Paediatrics' },
  { id: 'yousef', name: 'Dr. Yousef Saleh', specialty: 'Orthopaedics' },
];

export const DoctorPicker = () => {
  const [open, setOpen] = useState(false);
  const [doctorId, setDoctorId] = useState();
  const selected = DOCTORS.find((doctor) => doctor.id === doctorId);

  return (
    <Combobox open={open} onOpenChange={setOpen}>
      <Combobox.Trigger className="w-72">
        <Combobox.Value placeholder="Choose a doctor">{selected?.name}</Combobox.Value>
      </Combobox.Trigger>

      <Combobox.Content>
        <Combobox.Input placeholder="Search doctors…" />

        <Combobox.List>
          <Combobox.Empty>No doctors match.</Combobox.Empty>

          {DOCTORS.map((doctor) => (
            <Combobox.Item
              key={doctor.id}
              value={`${doctor.name} ${doctor.specialty}`}
              data-checked={doctor.id === doctorId || undefined}
              onSelect={() => {
                setDoctorId(doctor.id);
                setOpen(false);
              }}
            >
              <span className="flex min-w-0 flex-1 flex-col">
                <span className="text-fg">{doctor.name}</span>
                <span className="text-fg-muted text-xs">{doctor.specialty}</span>
              </span>

              <Conditional.If condition={doctor.id === doctorId}>
                <CheckIcon aria-hidden="true" className="text-accent size-4" />
              </Conditional.If>
            </Combobox.Item>
          ))}
        </Combobox.List>
      </Combobox.Content>
    </Combobox>
  );
};

## Single select

Click or focus the trigger, type to filter, use the arrow keys to move and Enter to choose; Escape closes without choosing. The search box gets focus when the list opens.

## Building blocks

Combobox is the low-level UI behind [FormCombobox](/docs/components/form-combobox) and [FormMultiCombobox](/docs/components/form-multi-combobox). Use those in forms; use Combobox directly for filters and toolbars.

## Accessibility

- The trigger is a button that opens a popover; the search input controls a `listbox` of options.
- Arrow keys move, Enter chooses, Escape closes and returns focus to the trigger.

## RTL notes

- The chevron sits at the end of the trigger; the list text aligns to the start.
