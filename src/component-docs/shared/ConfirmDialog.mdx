import { useState } from 'react';
import { Button, IconButton } from '@components/ui';
import { ConfirmDialog } from '@components/shared';
import { Trash2 } from 'lucide-react';

export const ControlledDemo = () => {
  const [open, setOpen] = useState(false);
  return (
    <>
      <Button variant="outline" onClick={() => setOpen(true)}>Submit claim</Button>
      <ConfirmDialog
        tone="warning"
        open={open}
        onOpenChange={setOpen}
        title="Submit with missing documents?"
        description="The insurer may reject claims without a referral letter."
        confirmLabel="Submit anyway"
        onConfirm={() => new Promise((resolve) => setTimeout(resolve, 1200))}
      />
    </>
  );
};

## With a trigger

Without `open`, the first child is the trigger. Return a promise from `onConfirm` and the button shows a spinner, then the dialog closes when it resolves (and stays open if it rejects).

<Example title="Delete">
  <ConfirmDialog
    tone="danger"
    title="Delete this visit?"
    description="The visit and its notes are removed for everyone. This can't be undone."
    confirmLabel="Delete visit"
    onConfirm={() => new Promise((resolve) => setTimeout(resolve, 1200))}
  >
    <IconButton variant="outline" tone="danger" aria-label="Delete visit">
      <Trash2 />
    </IconButton>
  </ConfirmDialog>
</Example>

## Controlled

With `open` and `onOpenChange`, open it from code, for example after a check fails. Children are then `[content?, actions?]` (no trigger).

## Tones

| `tone` | Icon | Confirm button | Cancel shown |
|---|---|---|---|
| `danger` (default) | ✕ circle | danger | yes |
| `warning` | triangle | neutral | yes |
| `success` | check circle | neutral | no |
| `info` | info | neutral | no |

Tone is carried by the icon and the confirm button; the dialog has no coloured border.

## Accessibility

- A modal dialog with the title as its name and the description wired to `aria-describedby`; focus returns to the trigger on close.
- Not dismissible by default, so a stray Escape can't skip the decision.

## RTL notes

- The icon sits at the start of the title and the buttons at the end of the footer, so both mirror in Arabic.
