import { useState } from 'react';
import { demoSource } from '@component-docs/blocks';
import { FileUploader } from '@components/shared';
import pageSource from './FileUploader.mdx?raw';

export const SingleDemo = () => {
  const [file, setFile] = useState(null);
  return (
    <div className="w-96">
      <FileUploader name="referral" value={file} onChange={setFile}>
        <FileUploader.DropzoneGroup>
          <FileUploader.Dropzone accept=".pdf,.doc,.docx" placeholder="Drop the referral letter or click to browse" />
          <FileUploader.Actions />
        </FileUploader.DropzoneGroup>
      </FileUploader>
    </div>
  );
};

export const GalleryDemo = () => {
  const [files, setFiles] = useState([]);
  return (
    <div className="w-96">
      <FileUploader name="photos" value={files} onChange={setFiles} accept="image/*" multiple>
        <FileUploader.DropzoneGroup>
          <FileUploader.Dropzone placeholder="Drop images" />
        </FileUploader.DropzoneGroup>
        <FileUploader.FileList variant="chips" />
        <FileUploader.ImagePreview />
      </FileUploader>
    </div>
  );
};

## Single file

`DropzoneGroup` holds the drop zone and, for one file, `Actions` (clear, optional download).

<Example title="One document" code={demoSource('SingleDemo', pageSource)}>
  <SingleDemo />
</Example>

## Multiple, with previews

With `multiple`, the value is `File[]`. Add `FileList` (list or chips) and `ImagePreview` below the zone.

<Example title="Images" code={demoSource('GalleryDemo', pageSource)}>
  <GalleryDemo />
</Example>

In forms, use [FormUploadFile](/docs/components/form-upload-file), which wires these parts to react-hook-form.

## Accessibility

- The drop zone wraps a native file input, so it is focusable and opens the file picker with Enter or Space.
- Clear, remove and download buttons have translated labels.

## RTL notes

- File sizes are isolated left to right; remove buttons sit at the end of each row, on the left in Arabic.
