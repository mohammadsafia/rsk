import { Conditional } from '@components/utils';

## Demo

<div className="space-y-2">
  <Conditional>
    <Conditional.If condition={true}>
      <div className="text-success">✓ Condition is true</div>
    </Conditional.If>
    <Conditional.Else>
      <div className="text-danger">✗ Condition is false</div>
    </Conditional.Else>
  </Conditional>

  <Conditional>
    <Conditional.If condition={false}>
      <div className="text-success">✓ This won't show</div>
    </Conditional.If>
    <Conditional.Else>
      <div className="text-fg-muted">→ Fallback content</div>
    </Conditional.Else>
  </Conditional>
</div>

## Imports

```tsx
import { Conditional } from '@components/utils';
```

## Code

```tsx
<Conditional>
  <Conditional.If condition={true}>Content</Conditional.If>
  <Conditional.Else>Fallback</Conditional.Else>
</Conditional>
```
