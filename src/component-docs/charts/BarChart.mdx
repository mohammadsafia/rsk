import { BarChart } from '@components/charts';

export const CLINICS = [
  { clinic: 'Abdoun', paid: 18200, pending: 2400 },
  { clinic: 'Khalda', paid: 14100, pending: 3900 },
  { clinic: 'Sweifieh', paid: 16800, pending: 1200 },
  { clinic: 'Zarqa', paid: 9800, pending: 2100 },
];

## Columns

<Example title="Revenue by clinic">
  <BarChart
    className="w-full"
    aria-label="Paid and pending revenue by clinic; Abdoun leads"
    data={CLINICS}
    series={[{ key: 'paid', label: 'Paid' }, { key: 'pending', label: 'Pending', tone: 'warning' }]}
    x={{ key: 'clinic' }}
    valueFormat="money"
  />
</Example>

## Horizontal, stacked

Horizontal bars suit long category names; `stacked` shows each total and its parts.

<Example title="Stacked bars">
  <BarChart
    className="w-full"
    aria-label="Total revenue by clinic, split into paid and pending"
    data={CLINICS}
    series={[{ key: 'paid', label: 'Paid' }, { key: 'pending', label: 'Pending', tone: 'warning' }]}
    x={{ key: 'clinic' }}
    valueFormat="money"
    orientation="horizontal"
    stacked
  />
</Example>

## Accessibility

- A `figure` named by `aria-label`; the legend's buttons toggle series with `aria-pressed`, and the last visible series can't be hidden.
- "View as table" swaps the chart for a table of the same values; arrow keys move between points (recharts accessibility layer).
- Animation is off under `prefers-reduced-motion`.

## RTL notes

- In Arabic columns run right to left and the value axis sits on the right; horizontal bars grow from the right.
