import { useState } from 'react';
import { SearchField } from '@blocks/search-field';

export const PEOPLE = [
  'Lina Khoury',
  'Omar Nasser',
  'Sara Haddad',
  'Yousef Saleh',
  'Layla Mansour',
  'Karim Aziz',
  'Nour Farah',
  'Rami Daher',
];

export const PeopleSearch = () => {
  const [chosen, setChosen] = useState('');

  const handleSearch = (query) => PEOPLE.filter((name) => name.toLowerCase().includes(query.toLowerCase()));

  return (
    <div className="flex w-80 max-w-full flex-col gap-3">
      <SearchField
        label="Find a colleague"
        placeholder="Type a name"
        onSearch={handleSearch}
        getLabel={(name) => name}
        getKey={(name) => name}
        onSelect={setChosen}
      />

      <p className="text-fg-muted text-sm">Picked: {chosen || 'nobody yet'}</p>
    </div>
  );
};

## Search as you type

Type two letters: the field waits for a pause (250 ms), calls `onSearch` and lists what it returns. ↑ and ↓ move through the suggestions, Enter picks one and Escape closes them (a second Escape clears the field).

## Add it

```bash
pnpm kit:add search-field
```

It lands in `src/blocks/search-field/` with its English and Arabic copy (`src/locales/{en,ar}/block-search-field.json`); `pnpm kit:update` keeps it current, and `pnpm kit:remove search-field` takes it out again.

The block never fetches on its own: `onSearch` is where your app calls its API.

## Accessibility

- The input is a `combobox` controlling a `listbox`; the highlighted suggestion is its `aria-activedescendant`, so focus never leaves the input.
- One live region announces “Searching…”, the number of results, “No results” or a failed search.

## RTL notes

- The search icon sits at the start of the field and the text aligns to the start, in both directions.
